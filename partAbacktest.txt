//@version=5
// =============================================================================
// Part A — SuperTrend Strategy (alias of v1 baseline)
// -----------------------------------------------------------------------------
// This file matches partAbacktest_v1.txt (pre CSV-align update).
// For TP35% / SL10% / $1M defaults targeting >87% on BTCUSD, use:
//   partAbacktest_v3.txt
// Enhanced filter experiment: partAbacktest_v2.txt
//
// Based on SuperTrend_indicator.txt signal logic (Buy/Sell on trend flip).
// How to use in TradingView:
//   1. Pine Editor → Open → New blank strategy
//   2. Paste this script → Save → Add to chart
//   3. Strategy Tester: set range from 2023-01-01 (script also filters dates)
//   4. Pick a symbol/timeframe (e.g. BTCUSDT.P 1H) and tune inputs
//
// Defaults (v1):
//   ATR Period=10, ATR Multiplier=3.0, TP %=3.0, SL %=1.5, capital=10,000
//   Enable Long=true, Enable Short=true, ATR method=true (RMA atr)
// =============================================================================
strategy(
     title              = "Part A SuperTrend Strategy",
     overlay            = true,
     format             = format.price,
     precision          = 2,
     initial_capital    = 10000,
     default_qty_type   = strategy.percent_of_equity,
     default_qty_value  = 100,
     pyramiding         = 0,
     commission_type    = strategy.commission.percent,
     commission_value   = 0.04,
     slippage           = 2,
     process_orders_on_close = true,
     calc_on_every_tick = false)

// --- Inputs: SuperTrend (match reference indicator) -------------------------
atrPeriod   = input.int(10,    "ATR Period", minval=1, group="SuperTrend")
src         = input.source(hl2, "Source", group="SuperTrend")
multiplier  = input.float(3.0, "ATR Multiplier", step=0.1, minval=0.1, group="SuperTrend")
changeATR   = input.bool(true, "Use RMA ATR (uncheck = SMA of TR)", group="SuperTrend")

// --- Inputs: Risk management ------------------------------------------------
tpPerc      = input.float(3.0, "Take Profit %", step=0.1, minval=0.1, group="Risk")
slPerc      = input.float(1.5, "Stop Loss %",   step=0.1, minval=0.1, group="Risk")
enableLong  = input.bool(true,  "Enable Long",  group="Trade Direction")
enableShort = input.bool(true,  "Enable Short", group="Trade Direction")

// --- Inputs: Display --------------------------------------------------------
showSignals   = input.bool(true, "Show Buy/Sell Signals", group="Display")
highlighting  = input.bool(true, "Highlighter On/Off",    group="Display")

// --- Inputs: Backtest window ------------------------------------------------
useDateFilter = input.bool(true, "Limit backtest to date range", group="Backtest")
startDate     = input.time(timestamp("2023-01-01 00:00"), "Start Date", group="Backtest")
endDate       = input.time(timestamp("2099-12-31 23:59"), "End Date",   group="Backtest")
inDateRange   = not useDateFilter or (time >= startDate and time <= endDate)

// --- SuperTrend core (equivalent to SuperTrend_indicator.txt) ---------------
atr2 = ta.sma(ta.tr(true), atrPeriod)
atr  = changeATR ? ta.atr(atrPeriod) : atr2

up  = src - multiplier * atr
up1 = nz(up[1], up)
up := close[1] > up1 ? math.max(up, up1) : up

dn  = src + multiplier * atr
dn1 = nz(dn[1], dn)
dn := close[1] < dn1 ? math.min(dn, dn1) : dn

var int trend = 1
trend := nz(trend[1], trend)
trend := trend == -1 and close > dn1 ? 1 : trend == 1 and close < up1 ? -1 : trend

buySignal  = trend == 1 and trend[1] == -1
sellSignal = trend == -1 and trend[1] == 1

// --- Entries / exits --------------------------------------------------------
// Long: enter on Buy; exit on TP, SL, or Sell (whichever first)
// Short: enter on Sell; exit on TP, SL, or Buy (whichever first)

longTP  = strategy.position_avg_price * (1 + tpPerc / 100.0)
longSL  = strategy.position_avg_price * (1 - slPerc / 100.0)
shortTP = strategy.position_avg_price * (1 - tpPerc / 100.0)
shortSL = strategy.position_avg_price * (1 + slPerc / 100.0)

if inDateRange
    if enableLong and buySignal
        // Opposite-signal exit for shorts is handled by strategy.entry flip,
        // but we also close explicitly so Sell always flattens a long.
        strategy.close("Short", comment="Buy signal — close short")
        strategy.entry("Long", strategy.long, comment="Buy")

    if enableShort and sellSignal
        strategy.close("Long", comment="Sell signal — close long")
        strategy.entry("Short", strategy.short, comment="Sell")

// Fixed TP / SL while in position (checked every bar in date range)
if inDateRange and strategy.position_size > 0
    strategy.exit("Long TP/SL", from_entry="Long", limit=longTP, stop=longSL, comment_profit="Long TP", comment_loss="Long SL")

if inDateRange and strategy.position_size < 0
    strategy.exit("Short TP/SL", from_entry="Short", limit=shortTP, stop=shortSL, comment_profit="Short TP", comment_loss="Short SL")

// Flatten if we leave the backtest window
if useDateFilter and not inDateRange and strategy.position_size != 0
    strategy.close_all(comment="End of backtest window")

// --- Plots (visual parity with indicator) -----------------------------------
upPlot = plot(trend == 1 ? up : na, title="Up Trend", style=plot.style_linebr, linewidth=2, color=color.green)
dnPlot = plot(trend == 1 ? na : dn, title="Down Trend", style=plot.style_linebr, linewidth=2, color=color.red)

plotshape(buySignal and showSignals ? up : na, title="Buy", text="Buy",
     location=location.absolute, style=shape.labelup, size=size.tiny,
     color=color.green, textcolor=color.white)
plotshape(sellSignal and showSignals ? dn : na, title="Sell", text="Sell",
     location=location.absolute, style=shape.labeldown, size=size.tiny,
     color=color.red, textcolor=color.white)

mPlot = plot(ohlc4, title="", display=display.none)
longFillColor  = highlighting ? (trend == 1 ? color.new(color.green, 90) : na) : na
shortFillColor = highlighting ? (trend == -1 ? color.new(color.red, 90) : na) : na
fill(mPlot, upPlot, title="UpTrend Highlighter", color=longFillColor)
fill(mPlot, dnPlot, title="DownTrend Highlighter", color=shortFillColor)

// --- Optional alerts --------------------------------------------------------
alertcondition(buySignal,  title="SuperTrend Buy",  message="SuperTrend Buy!")
alertcondition(sellSignal, title="SuperTrend Sell", message="SuperTrend Sell!")
